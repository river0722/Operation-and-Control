CLOSE
OPEN PLC 3 CLEAR
IF(P801=1)AND(M122=0)//向左走//801=1左限位为1避免在负限位还往负走
	I113=0
	I114=0
	I122=32
	COMMAND"#1J-"  
	P801=2
ELSE   
	IF(P801=2)    
		IF(M120=0)   //到达home标志
			COMMAND"#1J/"  //停止
			P801=3
		ENDIF
		IF(M122=1)   //到达负限位往正走偏置量
			P801=-1
			M172=200000
			COMMAND"#1J^*"//向正方向运动200000脉冲
			P813=0
		ENDIF
	ENDIF
	IF(P801=3)AND(M140=1) //P801=3并且未处于零位
		I123=-1   //设置回零运动方向为负
		I7013=0  //零点标志回零
		I7012=15    //组合方式，当I711=3代表的正限位有低信号
		COMMAND"#1HOME"   //开启寻找零点运动
		P801=4
	ENDIF
	IF(P801=4)AND(M162<4000)AND(M162>-4000) //电机位置在4000～-4000之间时
		P801=5
	ENDIF
	IF(P801=5)AND(M117=1)   //1号编码器捕捉信号
		COMMAND"#1HOMEZ"//将目前位置设置为零位
		P801=6
		I113=884000   //软正限位置为884000
		I114=-65000   //软正限位置为-6500
	ENDIF
	IF(P801=-1) //在负限位和零点
		IF(M140=0) //若到负限位信息为零
			P813=3
		ENDIF
		IF(M140=1)AND(P813=3)
			P801=1 //回到1状态
		ENDIF
	ENDIF
ENDIF

IF(P802=1)AND(M222=0)//向左走//802=1左限位为1避免在负限位还往负走
	I213=0
	I214=0
	I222=32
	COMMAND"#2J-"  
	P802=2
ELSE   
	IF(P802=2)    
		IF(M220=1)   //到达home标志
			COMMAND"#2J/"  //停止
			P802=3
		ENDIF
		IF(M222=1)   //到达负限位往正走偏置量
			P802=-1
			M272=49152
			COMMAND"#2J^*"//向正方向运动200000脉冲
			P823=0
		ENDIF
	ENDIF
	IF(P802=3)AND(M240=1) //P801=3并且未处于零位
		I223=-1   //设置回零运动方向为负
		I7023=0  //零点标志回零
		I7022=3    //组合方式，在索引为高和标志n为高时捕捉 I7mn2
		COMMAND"#2HOME"   //开启寻找零点运动
		P802=4
	ENDIF
	IF(P802=4)AND(M262<3000)AND(M262>-3000) //电机位置在4000～-4000之间时
		P802=5
	ENDIF
	IF(P802=5)AND(M217=0)   //1号编码器捕捉信号
		COMMAND"#2HOMEZ"//将目前位置设置为零位
		P802=6
		I213=884000   //软正限位置为884000
		I214=-65000   //软正限位置为-6500
	ENDIF
	IF(P802=-1) //在负限位和零点
		IF(M240=0) //若到负限位信息为零
			P823=3
		ENDIF
		IF(M240=1)AND(P823=3)
			P802=1 //回到1状态
		ENDIF
	ENDIF
ENDIF
